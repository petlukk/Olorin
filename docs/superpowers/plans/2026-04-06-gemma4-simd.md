```ea
// vec_ops.ea — Element-wise f32 vector operations (x86 AVX2)
//
// vec_add_f32:   out[i] = a[i] + b[i]
// vec_scale_f32: out[i] = a[i] * s
// vec_fma_f32:   out[i] = (a[i] + b[i]) * s
// vec_acc_f32:   out[i] += a[i] * s      (accumulate scaled)

#[cfg(x86_64)]

export func vec_add_f32(a: *restrict f32, b: *restrict f32, out: *mut f32, n: i32) {
    let mut i: i32 = 0
    while i + 8 <= n {
        let va: f32x8 = load(a, i)
        let vb: f32x8 = load(b, i)
        store(out, i, va .+ vb)
        i = i + 8
    }
    while i + 4 <= n {
        let va: f32x4 = load(a, i)
        let vb: f32x4 = load(b, i)
        store(out, i, va .+ vb)
        i = i + 4
    }
    while i < n {
        out[i] = a[i] + b[i]
        i = i + 1
    }
}

export func vec_scale_f32(a: *restrict f32, out: *mut f32, s: f32, n: i32) {
    let vs: f32x8 = splat(s)
    let mut i: i32 = 0
    while i + 8 <= n {
        store(out, i, load(a, i) .* vs)
        i = i + 8
    }
    let vs4: f32x4 = splat(s)
    while i + 4 <= n {
        store(out, i, load(a, i) .* vs4)
        i = i + 4
    }
    while i < n {
        out[i] = a[i] * s
        i = i + 1
    }
}

export func vec_fma_f32(a: *restrict f32, b: *restrict f32, out: *mut f32, s: f32, n: i32) {
    let vs: f32x8 = splat(s)
    let mut i: i32 = 0
    while i + 8 <= n {
        let va: f32x8 = load(a, i)
        let vb: f32x8 = load(b, i)
        store(out, i, (va .+ vb) .* vs)
        i = i + 8
    }
    let vs4: f32x4 = splat(s)
    while i + 4 <= n {
        let va: f32x4 = load(a, i)
        let vb: f32x4 = load(b, i)
        store(out, i, (va .+ vb) .* vs4)
        i = i + 4
    }
    while i < n {
        out[i] = (a[i] + b[i]) * s
        i = i + 1
    }
}

export func vec_acc_f32(out: *mut f32, a: *restrict f32, s: f32, n: i32) {
    let vs: f32x8 = splat(s)
    let mut i: i32 = 0
    while i + 8 <= n {
        let vo: f32x8 = load(out, i)
        let va: f32x8 = load(a, i)
        store(out, i, fma(va, vs, vo))
        i = i + 8
    }
    let vs4: f32x4 = splat(s)
    while i + 4 <= n {
        let vo: f32x4 = load(out, i)
        let va: f32x4 = load(a, i)
        store(out, i, fma(va, vs4, vo))
        i = i + 4
    }
    while i < n {
        out[i] = out[i] + a[i] * s
        i = i + 1
    }
}
```

```ea
// attn_ops.ea — Attention f32 dot product + weighted V accumulation (x86 AVX2)
//
// f32_dot:     returns dot(a, b) over n elements
// f32_dot_acc: out[i] += a[i] * s  (scaled accumulate, for V weighted sum)

#[cfg(x86_64)]

export func f32_dot(a: *restrict f32, b: *restrict f32, n: i32) -> f32 {
    let mut acc0: f32x8 = splat(0.0)
    let mut acc1: f32x8 = splat(0.0)
    let mut i: i32 = 0

    while i + 16 <= n {
        let va0: f32x8 = load(a, i)
        let vb0: f32x8 = load(b, i)
        let va1: f32x8 = load(a, i + 8)
        let vb1: f32x8 = load(b, i + 8)
        acc0 = fma(va0, vb0, acc0)
        acc1 = fma(va1, vb1, acc1)
        i = i + 16
    }

    while i + 8 <= n {
        acc0 = fma(load(a, i), load(b, i), acc0)
        i = i + 8
    }

    let mut result: f32 = reduce_add(acc0) + reduce_add(acc1)

    while i + 4 <= n {
        let va: f32x4 = load(a, i)
        let vb: f32x4 = load(b, i)
        result = result + reduce_add(va .* vb)
        i = i + 4
    }

    while i < n {
        result = result + a[i] * b[i]
        i = i + 1
    }
    return result
}

export func f32_dot_acc(out: *mut f32, a: *restrict f32, s: f32, n: i32) {
    let vs: f32x8 = splat(s)
    let mut i: i32 = 0

    while i + 16 <= n {
        let vo0: f32x8 = load(out, i)
        let va0: f32x8 = load(a, i)
        let vo1: f32x8 = load(out, i + 8)
        let va1: f32x8 = load(a, i + 8)
        store(out, i, fma(va0, vs, vo0))
        store(out, i + 8, fma(va1, vs, vo1))
        i = i + 16
    }

    while i + 8 <= n {
        store(out, i, fma(load(a, i), vs, load(out, i)))
        i = i + 8
    }

    let vs4: f32x4 = splat(s)
    while i + 4 <= n {
        let vo: f32x4 = load(out, i)
        store(out, i, fma(load(a, i), vs4, vo))
        i = i + 4
    }

    while i < n {
        out[i] = out[i] + a[i] * s
        i = i + 1
    }
}
```

```ea
// bare_rmsnorm.ea — RMSNorm without weight (for V normalization)
//
// x[i] = x[i] * rsqrt(mean(x^2) + eps)
// Same as gemma4_rmsnorm but no weight array.

#[cfg(x86_64)]

export func bare_rmsnorm_f32(x: *mut f32, n: i32, eps: f32) {
    // Pass 1: Sum of squares
    let mut acc0: f32x8 = splat(0.0)
    let mut acc1: f32x8 = splat(0.0)
    let mut i: i32 = 0

    while i + 16 <= n {
        let v0: f32x8 = load(x, i)
        let v1: f32x8 = load(x, i + 8)
        acc0 = fma(v0, v0, acc0)
        acc1 = fma(v1, v1, acc1)
        i = i + 16
    }

    while i + 8 <= n {
        let v: f32x8 = load(x, i)
        acc0 = fma(v, v, acc0)
        i = i + 8
    }

    let mut ss: f32 = reduce_add(acc0) + reduce_add(acc1)
    while i < n {
        ss = ss + x[i] * x[i]
        i = i + 1
    }

    let scale: f32 = rsqrt(ss / to_f32(n) + eps)

    // Pass 2: Scale in-place
    let vs: f32x8 = splat(scale)
    i = 0
    while i + 8 <= n {
        store(x, i, load(x, i) .* vs)
        i = i + 8
    }
    let vs4: f32x4 = splat(scale)
    while i + 4 <= n {
        store(x, i, load(x, i) .* vs4)
        i = i + 4
    }
    while i < n {
        x[i] = x[i] * scale
        i = i + 1
    }
}
```

Replaces `cap * tanh(x * inv_cap)` over vocab_size (256k elements). Uses `exp()` since `tanh` is not an Ea intrinsic: `tanh(x) = 1 - 2/(exp(2x) + 1)`.

- [ ] **Step 1: Write the Ea kernel**

Create `kernels/softcap.ea`:

```ea
// softcap.ea — Logit softcapping: out[i] = cap * tanh(x[i] / cap)
//
// tanh(x) = 1 - 2 / (exp(2x) + 1)
// Since no tanh intrinsic, built from exp().

#[cfg(x86_64)]

export func softcap_f32(data: *mut f32, n: i32, cap: f32) {
    let inv_cap: f32 = 1.0 / cap
    let v_inv_cap: f32x8 = splat(inv_cap)
    let v_cap: f32x8 = splat(cap)
    let v_two: f32x8 = splat(2.0)
    let v_one: f32x8 = splat(1.0)
    let mut i: i32 = 0

    while i + 8 <= n {
        let x: f32x8 = load(data, i) .* v_inv_cap
        let exp2x: f32x8 = exp(x .* v_two)
        let th: f32x8 = v_one .- v_two ./ (exp2x .+ v_one)
        store(data, i, th .* v_cap)
        i = i + 8
    }

    while i < n {
        let x: f32 = data[i] * inv_cap
        let exp2x: f32 = exp(x * 2.0)
        let th: f32 = 1.0 - 2.0 / (exp2x + 1.0)
        data[i] = th * cap
        i = i + 1
    }
}
```
