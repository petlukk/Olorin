> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement Per-Layer Embeddings (PLE) for Gemma 4 E2B to produce correct output.

**Architecture:** Two-phase PLE: Phase A computes per-layer signals once per token (BF16 matvec + Q6K lookup + RMSNorm). Phase B applies gated bottleneck adapter per layer (Q4K matvecs + GELU + residual). New `bf16_matvec.ea` kernel for BF16 × f32 dot product.

**Tech Stack:** Eä SIMD kernels, existing Q4K/Q6K infrastructure, Rust FFI wrappers.

- [ ] **Step 1: Write `kernels/bf16_matvec.ea`**

BF16→f32 conversion: shift left 16 bits. 4-row batched for throughput.

```
// bf16_matvec.ea — BF16 weight × f32 input dot product
//
// BF16→f32: reinterpret (u16 as u32) << 16 as f32.
// 4-element f32x4 SIMD, 4-row batched variant.

#[cfg(x86_64)]

// Single-row dot: BF16 weight row × f32 input → f32 scalar
export func bf16_dot_f32(
    weight: *restrict u16, input: *restrict f32, n_cols: i32
) -> f32 {
    let mut acc0: f32x4 = splat(0.0)
    let mut acc1: f32x4 = splat(0.0)
    let mut i: i32 = 0

    while i + 8 <= n_cols {
        // Load 8 BF16 values, convert to f32
        let w0_raw: i32 = to_i32(weight[i]) << 16
        let w1_raw: i32 = to_i32(weight[i+1]) << 16
        let w2_raw: i32 = to_i32(weight[i+2]) << 16
        let w3_raw: i32 = to_i32(weight[i+3]) << 16
        let w0: f32x4 = [bitcast_f32(w0_raw), bitcast_f32(w1_raw), bitcast_f32(w2_raw), bitcast_f32(w3_raw)]

        let w4_raw: i32 = to_i32(weight[i+4]) << 16
        let w5_raw: i32 = to_i32(weight[i+5]) << 16
        let w6_raw: i32 = to_i32(weight[i+6]) << 16
        let w7_raw: i32 = to_i32(weight[i+7]) << 16
        let w1v: f32x4 = [bitcast_f32(w4_raw), bitcast_f32(w5_raw), bitcast_f32(w6_raw), bitcast_f32(w7_raw)]

        let in0: f32x4 = load(input, i)
        let in1: f32x4 = load(input, i + 4)

        acc0 = fma(w0, in0, acc0)
        acc1 = fma(w1v, in1, acc1)
        i = i + 8
    }

    let mut result: f32 = reduce_add(acc0) + reduce_add(acc1)

    while i < n_cols {
        let w_raw: i32 = to_i32(weight[i]) << 16
        result = result + bitcast_f32(w_raw) * input[i]
        i = i + 1
    }

    return result
}

// 4-row batched: compute 4 dot products simultaneously
export func bf16_dot_f32_4row(
    w0: *restrict u16, w1: *restrict u16, w2: *restrict u16, w3: *restrict u16,
    input: *restrict f32,
    out scores: *mut f32 [cap: 4],
    n_cols: i32
) {
    scores[0] = bf16_dot_f32(w0, input, n_cols)
    scores[1] = bf16_dot_f32(w1, input, n_cols)
    scores[2] = bf16_dot_f32(w2, input, n_cols)
    scores[3] = bf16_dot_f32(w3, input, n_cols)
}
```

Note: Check eacompute for `bitcast_f32` intrinsic. If not available, check for alternative like `reinterpret_f32` or `as_f32`. Search `src/typeck/intrinsics*.rs`, `src/codegen/simd*.rs`, `CHANGELOG.md`, `README.md`, and `tests/` in the eacompute repo.

```rust
// ---------------------------------------------------------------------------
// BF16 matrix-vector multiply
// ---------------------------------------------------------------------------

/// BF16 matrix-vector: weight (n_rows × n_cols, BF16) × input (f32) → output (f32).
///
/// BF16 weight data is 2 bytes per element. Row stride = n_cols * 2 bytes.
pub fn bf16_matvec(
    weight: *const u8,
    input: &[f32],
    output: &mut [f32],
    n_rows: usize,
    n_cols: usize,
) {
    let row_bytes = n_cols * 2; // BF16 = 2 bytes per element

    let full_quads = n_rows / 4;
    let remainder = n_rows % 4;

    unsafe {
        for quad in 0..full_quads {
            let base_row = quad * 4;
            let w0 = weight.add(base_row * row_bytes) as *const u16;
            let w1 = weight.add((base_row + 1) * row_bytes) as *const u16;
            let w2 = weight.add((base_row + 2) * row_bytes) as *const u16;
            let w3 = weight.add((base_row + 3) * row_bytes) as *const u16;

            ffi_inference::bf16_dot_f32_4row(
                w0, w1, w2, w3,
                input.as_ptr(),
                output.as_mut_ptr().add(base_row),
                n_cols as i32,
            );
        }

        for i in 0..remainder {
            let row = full_quads * 4 + i;
            let w = weight.add(row * row_bytes) as *const u16;
            output[row] = ffi_inference::bf16_dot_f32(
                w, input.as_ptr(), n_cols as i32,
            );
        }
    }
}
```

PLE token embeddings have row size `ple_dim * n_layers` (8960) instead of `hidden_dim` (1536). Add to `src/inference/dequant.rs`:

```rust
/// Dequantize a row from Q6K PLE token embedding table.
///
/// Same as q6k_embed_lookup but with configurable row width.
/// PLE embeddings have row_dim = ple_dim * n_layers.
pub fn q6k_dequant_row(
    weight: *const u8,
    row_id: usize,
    output: &mut [f32],
    row_dim: usize,
) {
    debug_assert!(row_dim % Q6K_BLOCK_SIZE == 0);
    let n_blocks = row_dim / Q6K_BLOCK_SIZE;
    let row_bytes = n_blocks * Q6K_BLOCK_BYTES;
    let row_base = unsafe { weight.add(row_id * row_bytes) };

    for blk in 0..n_blocks {
        let block = unsafe { row_base.add(blk * Q6K_BLOCK_BYTES) };
        let out_base = blk * Q6K_BLOCK_SIZE;

        let d_raw = unsafe { u16::from_le_bytes([*block.add(208), *block.add(209)]) };
        let d = f16_to_f32_scalar(d_raw);

        for half in 0..2usize {
            let ql_base = unsafe { block.add(half * 64) };
            let qh_base = unsafe { block.add(128 + half * 32) };
            let sc_base = unsafe { block.add(192 + half * 8) };
            let out_half = out_base + half * 128;

            for group in 0..4usize {
                let ql_off = if group < 2 { 0 } else { 32 };
                let use_high_nibble = group % 2 == 1;
                let qh_shift = group * 2;
                let sc_idx = group * 2;

                for sub in 0..2usize {
                    let sc = unsafe { *sc_base.add(sc_idx + sub) as i8 } as f32;
                    let out_off = out_half + group * 32 + sub * 16;

                    for l in 0..16usize {
                        let ql_byte = unsafe { *ql_base.add(ql_off + sub * 16 + l) };
                        let qh_byte = unsafe { *qh_base.add(sub * 16 + l) };

                        let low = if use_high_nibble { ql_byte >> 4 } else { ql_byte & 0xF };
                        let high = (qh_byte >> qh_shift) & 3;
                        let q6 = (low | (high << 4)) as i8 as f32 - 32.0;

                        output[out_off + l] = d * sc * q6;
                    }
                }
            }
        }
    }
}
```

```rust
/// Phase A: compute per-layer PLE signal for this token.
/// Called once per token before the layer loop.
pub(crate) fn prepare_ple(
    &mut self,
    model: &Gemma4Model,
    token_id: u32,
) {
    let ple_dim = model.ple_dim;
    if ple_dim == 0 || model.ple_token_embd.is_null() {
        return;
    }
    let n_layers = model.n_layers;
    let hd = model.hidden_dim;
    let total = ple_dim * n_layers; // 8960

    // 1. Q6K dequant lookup: ple_token_embd[token_id] -> raw_ple[total]
    //    Scale by sqrt(ple_dim)
    dequant::q6k_dequant_row(
        model.ple_token_embd,
        token_id as usize,
        &mut self.ple_signal,
        total,
    );
    let embd_scale = (ple_dim as f32).sqrt();
    for v in self.ple_signal[..total].iter_mut() {
        *v *= embd_scale;
    }

    // 2. BF16 matvec: ple_model_proj @ scaled_embedding -> proj_ple[total]
    //    Scale by 1/sqrt(hidden_dim)
    //    Reuse ple_out (1536) won't work — need full total (8960). Use ple_gate? No, too small.
    //    Use a temporary or in-place: compute into ple_out extended? No.
    //    Actually: compute BF16 matvec into a scratch, then add.
    //    We can compute into ple_gate... no, it's ple_dim=256.
    //    Simplest: allocate a local Vec for proj result, or reuse ple_signal in two passes.
    //
    //    Better approach: compute proj into ple_out (reused as scratch, wrong size).
    //    Nope — ple_out is hidden_dim=1536, but we need 8960.
    //
    //    Solution: compute BF16 matvec directly, then fuse add+scale in place.
    //    We need a scratch buffer of size total=8960 for the projection result.
    //    Add ple_proj_scratch to Gemma4State.
    //
    //    For now, allocate locally — this runs once per token.
    let mut proj_ple = vec![0.0f32; total];
    matmul::bf16_matvec(
        model.ple_model_proj,
        &self.x[..hd], // scaled embedding
        &mut proj_ple,
        total,
        hd,
    );
    let proj_scale = 1.0 / (hd as f32).sqrt();
    for v in proj_ple.iter_mut() {
        *v *= proj_scale;
    }

    // 3. RMSNorm each [ple_dim] slice of proj_ple with ple_proj_norm
    if !model.ple_proj_norm.is_null() {
        for il in 0..n_layers {
            let off = il * ple_dim;
            ffi_inference::gemma4_rmsnorm(
                proj_ple[off..].as_ptr(),
                model.ple_proj_norm,
                proj_ple[off..].as_mut_ptr(),
                ple_dim as i32,
                model.rms_eps,
            );
        }
    }

    // 4. Add + scale: ple_signal = (proj_ple + raw_ple) / sqrt(2)
    let inv_sqrt2 = 1.0 / 2.0f32.sqrt();
    for i in 0..total {
        self.ple_signal[i] = (self.ple_signal[i] + proj_ple[i]) * inv_sqrt2;
    }
}
```
